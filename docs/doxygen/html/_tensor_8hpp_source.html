76inline auto check_cuda(cudaError_t status, const char* file, int line) -> void
77{
78 if (status != cudaSuccess)
79 {
80 const std::string message = std::string("CUDA error at ") + file + ":" + std::to_string(line) + ": "
81 + cudaGetErrorString(status);
