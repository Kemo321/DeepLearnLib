58{
59 unsigned long long hash = seed + (static_cast<unsigned long long>(index) + 1ULL) * 0x9E3779B97F4A7C15ULL;

87inline auto next_bounded(std::mt19937& rng, std::uint32_t bound) -> std::uint32_t
88{
89 if (bound <= 1U)
90 {
91 return 0U;
92 }
93 const std::uint32_t limit = (std::numeric_limits<std::uint32_t>::max() / bound) * bound;
94 std::uint32_t draw = 0U;
95 do
96 {
97 draw = static_cast<std::uint32_t>(rng());
98 } while (draw >= limit);
99 return draw % bound;
100}
