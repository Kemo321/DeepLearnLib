80inline auto next_bounded(std::mt19937& rng, std::uint32_t bound) -> std::uint32_t
81{
82 if (bound <= 1U)
83 {
84 return 0U;
85 }
86 const std::uint32_t limit = (std::numeric_limits<std::uint32_t>::max() / bound) * bound;
87 std::uint32_t draw = 0U;
88 do
89 {
90 draw = static_cast<std::uint32_t>(rng());
91 } while (draw >= limit);
92 return draw % bound;
93}
