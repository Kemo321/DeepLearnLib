30inline auto check_cudnn(cudnnStatus_t status, const char* file, int line) -> void
31{
32 if (status != CUDNN_STATUS_SUCCESS)
33 {
34 const std::string message = std::string("cuDNN error at ") + file + ":" + std::to_string(line) + ": "
35 + cudnnGetErrorString(status);
36 log_error_message(message);
37 throw std::runtime_error(message);
38 }
39}
