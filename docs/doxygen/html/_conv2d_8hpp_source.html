31inline auto check_cudnn(cudnnStatus_t status, const char* file, int line) -> void
32{
33 if (status != CUDNN_STATUS_SUCCESS)
34 {
35 const std::string message = std::string("cuDNN error at ") + file + ":" + std::to_string(line) + ": "
36 + cudnnGetErrorString(status);
37 log_error_message(message);
38 throw std::runtime_error(message);
39 }
40}
